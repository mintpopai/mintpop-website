# mintpop-website 中英文 i18n 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 官网支持简体中文与英文两套预渲染页面：英文在根路径不变，中文加 `/zh` 前缀，SEO（hreflang / og:locale / sitemap / `<html lang>`）按语言完整输出。

**Architecture:** 语言由 URL 路径唯一决定：路由 `meta.locale` → router 守卫设置 vue-i18n 全局 locale → 组件文案；同一 `meta.locale` 也驱动 `useSeo` 的本地化 head。路由表、SEO 配置均从单一 base 定义生成两套（英文树 + `/zh` 中文树），vite-ssg 自动把两套路由都预渲染成静态页。

**Tech Stack:** Vue 3（`<script setup>` + TS）、vue-router 4、vue-i18n（v11，`legacy: false`）、vite-ssg、@unhead/vue v2、Vitest。

**Spec:** `docs/superpowers/specs/2026-07-03-i18n-design.md`

**设计文档：** `docs/superpowers/specs/2026-07-03-i18n-design.md`（本计划的需求与决策依据）。

```ts
/**
 * 语言常量：成员名按全局规范用 SCREAMING_SNAKE_CASE；
 * 取值是小写 BCP-47 码（'en'/'zh'）——要进 URL、<html lang> 与 vue-i18n API，
 * 均为外部标准要求小写，按「技术符号例外」处理（见 i18n 设计文档）。
 */
export const Locale = {
  EN: 'en',
  ZH: 'zh',
} as const

export type Locale = (typeof Locale)[keyof typeof Locale]

/** 中文路由前缀：语言由 URL 唯一决定的锚点 */
export const ZH_PREFIX = '/zh'

/** 语言切换器的显示标签：展示目标语言自己的名字 */
export const LOCALE_LABELS: Record<Locale, string> = {
  [Locale.EN]: 'EN',
  [Locale.ZH]: '中文',
}

/** 把「无语言前缀的基准路径」本地化成目标语言路径：'/legal'+zh → '/zh/legal'；'/'+zh → '/zh' */
export function localizePath(path: string, locale: Locale): string {
  if (locale === Locale.EN) {
    return path
  }
  return path === '/' ? ZH_PREFIX : `${ZH_PREFIX}${path}`
}

/** 去掉语言前缀得到基准路径：'/zh/legal' → '/legal'；'/zh' → '/'；前缀须整段匹配（'/zhang' 原样返回） */
export function stripLocale(path: string): string {
  if (path === ZH_PREFIX || path === `${ZH_PREFIX}/`) {
    return '/'
  }
  if (path.startsWith(`${ZH_PREFIX}/`)) {
    return path.slice(ZH_PREFIX.length)
  }
  return path
}
```

```ts
declare module 'vue-router' {
  interface RouteMeta {

/** 中文树路由 name 后缀：HOME → HOME_ZH */

// base 路由表：path 恒为英文基准路径（与 SEO 配置同源），双语树由它生成。
// zh=false 的条目不进中文树（404 页只预渲染英文一份，nginx error_page 只指向一个文件）。
interface BaseRouteDef {
  path: string
  name: SeoRouteKey
  // 若 RouteRecordRaw['component'] 索引类型在当前 vue-router 版本下报错，
  // 改用 vue-router 导出的 RawRouteComponent 类型，语义等价
```

```ts
import { useHead } from '@unhead/vue'
import { useRoute } from 'vue-router'

import { Locale, localizePath } from '@/config/locale'
import { SEO, type SeoRouteKey } from '@/config/seo'
import { absoluteUrl, OG_IMAGE_PATH, SITE_NAME } from '@/config/site'

// OG 图标准社交预览尺寸
const OG_IMAGE_WIDTH = '1200'
const OG_IMAGE_HEIGHT = '630'

// <html lang> 与 og:locale 的语言映射
const HTML_LANG: Record<Locale, string> = {
  [Locale.EN]: 'en',
  [Locale.ZH]: 'zh-CN',
}
const OG_LOCALE: Record<Locale, string> = {
  [Locale.EN]: 'en_US',
  [Locale.ZH]: 'zh_CN',
}

interface SeoMeta {
  name?: string
  property?: string
  content: string
}

interface SeoLink {
  rel: string
  hreflang?: string
  href: string
}

export interface SeoHead {
  title: string
  htmlAttrs: { lang: string }
  meta: SeoMeta[]
  link: SeoLink[]
}

/**
 * 组装某路由某语言的完整 head 声明（纯函数，便于单测）：
 * 本地化 title/描述/OG/Twitter + canonical + <html lang>；
 * 可收录页互挂 hreflang（en/zh/x-default，x-default 指英文），noindex 页输出 robots。
 */
export function buildSeoHead(key: SeoRouteKey, locale: Locale): SeoHead {
  const entry = SEO[key]
  const copy = entry.copy[locale]
  const canonical = absoluteUrl(localizePath(entry.path, locale))
  const ogImage = absoluteUrl(OG_IMAGE_PATH)

  const meta: SeoMeta[] = [
    { name: 'description', content: copy.description },
    // Open Graph
    { property: 'og:type', content: 'website' },
    { property: 'og:site_name', content: SITE_NAME },
    { property: 'og:locale', content: OG_LOCALE[locale] },
    { property: 'og:title', content: copy.title },
    { property: 'og:description', content: copy.description },
    { property: 'og:url', content: canonical },
    { property: 'og:image', content: ogImage },
    { property: 'og:image:width', content: OG_IMAGE_WIDTH },
    { property: 'og:image:height', content: OG_IMAGE_HEIGHT },
    { property: 'og:image:alt', content: copy.title },
    // Twitter
    { name: 'twitter:card', content: 'summary_large_image' },
    { name: 'twitter:title', content: copy.title },
    { name: 'twitter:description', content: copy.description },
    { name: 'twitter:image', content: ogImage },
    { name: 'twitter:image:alt', content: copy.title },
  ]

  const link: SeoLink[] = [{ rel: 'canonical', href: canonical }]
  if (entry.noindex) {
    meta.push({ name: 'robots', content: 'noindex' })
  } else {
    // hreflang 互链让搜索引擎知晓两个语言版本互为对照
    link.push(
      { rel: 'alternate', hreflang: 'en', href: absoluteUrl(localizePath(entry.path, Locale.EN)) },
      { rel: 'alternate', hreflang: 'zh', href: absoluteUrl(localizePath(entry.path, Locale.ZH)) },
      { rel: 'alternate', hreflang: 'x-default', href: absoluteUrl(entry.path) },
    )
  }

  return {
    title: copy.title,
    htmlAttrs: { lang: HTML_LANG[locale] },
    meta,
    link,
  }
}

/**
 * 按路由 key 声明该页 head。语言从当前路由 meta.locale 读取（URL 唯一决定语言），
 * 各 View 调用方式不变：useSeo('HOME')。
 */
export function useSeo(key: SeoRouteKey): void {
  const route = useRoute()
  useHead(buildSeoHead(key, route.meta.locale ?? Locale.EN))
}
```

创建 `src/composables/useLocale.ts`：

```ts
import { computed } from 'vue'
import { useRoute } from 'vue-router'

import { Locale, LOCALE_LABELS, localizePath, stripLocale } from '@/config/locale'

// 语言偏好持久化键：手动切换时写入；回访时 App.vue 用它把用户带回偏好语言
export const STORAGE_KEY_LOCALE = 'mint-locale'

/** 记住用户手动选择的语言（SSG 构建期无 localStorage，直接跳过） */
export function rememberLocale(locale: Locale): void {
  if (typeof localStorage === 'undefined') {
    return
  }
  localStorage.setItem(STORAGE_KEY_LOCALE, locale)
}

/** 读取语言偏好；无记录或值非法（历史脏数据）返回 null */
export function readLocalePreference(): Locale | null {
  if (typeof localStorage === 'undefined') {
    return null
  }
  const saved = localStorage.getItem(STORAGE_KEY_LOCALE)
  return saved === Locale.EN || saved === Locale.ZH ? saved : null
}

/**
 * 语言上下文 composable：语言唯一由当前路由 meta.locale 决定。
 * lp() 把站内基准路径本地化到当前语言（页内链接用）；
 * switchTarget 是当前页在另一语言下的对等路径（切换器用，保留锚点）。
 */
export function useLocale() {
  const route = useRoute()
  const locale = computed<Locale>(() => route.meta.locale ?? Locale.EN)
  const otherLocale = computed<Locale>(() => (locale.value === Locale.ZH ? Locale.EN : Locale.ZH))
  const otherLabel = computed(() => LOCALE_LABELS[otherLocale.value])
  const lp = (path: string): string => localizePath(path, locale.value)
  const switchTarget = computed(
    () => localizePath(stripLocale(route.path), otherLocale.value) + route.hash,
  )
  return { locale, otherLocale, otherLabel, lp, switchTarget }
}
```
